\textbf{Ejercicio 1 (15 puntos):} Determine y clasifique los puntos críticos de la siguiente función

\begin{align*}
    f(x,y) &= e^x(x^2-4x)+y^2-4y \\
\end{align*}



\textbf{Solución:} Calculamos el gradiente

\begin{align*}
    \nabla f &=(e^x(x^2-4x)+e^x(2x-4), 2y-4) \\
    &=(e^x(x^2-2x-4), 2y-4) \\
\end{align*}

Luego necesitamos que

\begin{align*}
    e^x(x^2-2x-4) &= 0 \\
    2y-4 &= 0
\end{align*}

De donde se deduce que  $y=2$ y $x^2-2x-4=0$, luego 

\begin{align*}
    x &= \frac{1 \pm \sqrt{5}}{2}
\end{align*}

Concluimos que los puntos críticos son $\displaystyle \frac{1 + \sqrt{5}}{2}$ y $\displaystyle \frac{1 - \sqrt{5}}{2}$, ahora calculamos el Hessiano

\begin{align}
    H = \begin{pmatrix}
        e^x(x^2-6) & 0 \\
        0 & 2
    \end{pmatrix}
\end{align}

Luego $D(x,y)= 2e^x(x^2-6)$ \\

-Tenemos que $D( \frac{1 + \sqrt{5}}{2},2) > 0$ y además $f_{xx}(\frac{1 + \sqrt{5}}{2},2) > 0$ por lo tanto es un mínim local\\

- Tenemos que $D( \frac{1 - \sqrt{5}}{2},2) < 0$ por lo tanto es un punto silla \\

\textbf{Pregunta 3 (15 puntos):} Determine el centro de masas de la lámina delimitada por la región circular $x^2+y^2=4$ y cuya función de densidad es $\rho(x,y) = y+2$ \\ \\

\textbf{Solución: }primero calculamos la masa usando coordenadas polares

\begin{align*}
    m &= \int_0^{2\pi}\int_0^2 (r\sin(\theta)+2)r drd\theta \\
     &= 2\int_0^{2\pi} r^2\sin(\theta)+2r d\theta \\
     &= \int_0^{2\pi} \frac{r^3}{3}\sin(\theta)+r^2\bigg|_0^{2} d\theta\\
     &= \int_0^{2\pi} \frac{8}{3}\sin(\theta)+4 d\theta\\
     &= \left( 4\theta -\frac{8}{3}\cos(\theta) \right) \bigg|_0^{2\pi} \\
     &= 8\pi
\end{align*}


Ahora

\begin{align*}
    \overline{x} &= \frac{1}{8\pi} \iint_R xy+2x dA \\
\end{align*}

Pero antes de hacer cualquier cálculo podemos notar que ambas funciones $xy$ y $2x$ son funciones impares respecto a $x$ en un círculo centrado en 0, se puede concluir inmediatamente que

\begin{align*}
    \overline{x} =0
\end{align*}

Aqunque también se puede hacer calculando directamente las integrales. Por otra parte


\begin{align*}
    \overline{y}  &= \frac{1}{8\pi} \iint_R y^2+2y dA
\end{align*}

Nuevamente la función $2y$ es impar y esa integral se anula, solo nos queda


\begin{align*}
    \overline{y}  &= \frac{1}{8\pi} \iint_R y^2dA \\
    &=  \frac{1}{8\pi}\int_0^{2\pi}\int_0^2 r^3\sin^2(\theta) dr d\theta \\
    &=  \frac{1}{8\pi}\int_0^2 \sin^2(\theta) \frac{2^4}{4} d\theta \\
    &= \frac{1}{8\pi} 4\left( \frac{\theta}{2}-\frac{\sin(2\theta)}{4}\right)\bigg|_0^{2\pi} \\
    &=  \frac{1}{8\pi}4\pi \\
    &= \frac{1}{2}
\end{align*}

Luego el centro de masas es $\displaystyle \left( 0 , \frac{1}{2}\right)$

\textbf{Ejercicio 4 (15 puntos):} Determine 

\begin{align*}
    \iiint_R x^2yz dV
\end{align*}

Donde $R$ es la región del primer octante limitado por el cilindro $x^2+y^2=1$, el plano $z=1$ y el plano $XY$









\begin{figure}[h]
    \centering
    \includegraphics[width=0.5\textwidth]{Controles/region.jpg}
    \label{fig:etiqueta}
\end{figure}  


\textbf{Solución:} Pasando a coordenadas cilíndricas, la integral triple queda de la forma

\begin{align*}
    \int_0^{\pi/2}\int_0^1\int_0^1 r^4\cos^2(\theta)\sin(\theta)z dzdrd\theta &= \frac{1}{2}\int_0^{\pi/2}\int_0^1 r^4\cos^2(\theta)\sin(\theta) drd\theta  \\
     &= \frac{1}{10}\int_0^{\pi/2} \cos^2(\theta)\sin(\theta) d\theta  \\
\end{align*}

Para calcular la última integral hacemos $u=\cos(\theta)$, luego $du=-\sin(\theta)d\theta$, luego

\begin{align*}
    \int_0^{\pi/2}\int_0^1\int_0^1 r^4\cos^2(\theta)\sin(\theta)z dzdrd\theta &= \frac{1}{10}\int_1^{0} -u^2 du  \\
    &= \frac{1}{10}\frac{1}{3} \\
    &= \frac{1}{30}
\end{align*}
