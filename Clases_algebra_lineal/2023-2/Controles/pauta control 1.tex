       \item  Considere las matrices $A,D,Q \in \mathcal{M}_3(\mathbb{R})$, donde $Q$ es invertible y además,  ́$ A = Q^{-1}DQ$. Demuestre, mediante propiedades de matrices y determinantes que:

\begin{enumerate}
    \item {\bf (2 puntos)} $A^3=Q^{-1}D^3Q$.
    \item {\bf (3 puntos)} $A^n=Q^{-1}D^nQ$ para todo $n\in \mathbb{N}$.
    \item {\bf (5 puntos)} $|A|=|D|$.
\end{enumerate}
\textbf{Soluci\'on:}
\begin{enumerate}
    \item {\bf (2 puntos)} $A^3=(Q^{-1}DQ)(Q^{-1}DQ)(Q^{-1}DQ)=(Q^{-1}D^3Q)$.
    \item {\bf (3 puntos)} $A^n=(Q^{-1}DQ)\cdots (Q^{-1}DQ)$ $n$ veces, luego $A^n=Q^{-1}D^nQ$.
    \item {\bf (5 puntos)} $|A|=|Q^{-1}DQ|=|Q^{-1}| |D| |Q|$, como $Q$ es invertible $|Q^{-1}|\neq 0$ y $|Q^{-1}|=\dfrac{1}{|Q|}$, por lo que $|A|=|Q^{-1}DQ|=\dfrac{1}{|Q|} |D| |Q|=|D|$.
\end{enumerate}

    \item {\bf (15 puntos)} Usando el teorema del rango, determine los valores de $\alpha$ de modo que el sistema
    \begin{align*}
x + y +z &= 4 \\
2x + 5y-\alpha z &= 3\\
x + 2y +  z &= 8
\end{align*}

    tenga solución única, infinitas soluciones o ninguna solución. Para $\alpha=3$, determine el conjunto soluci\'on del sistema.
        
        \textbf{Soluci\'on}
Escribimos la matriz ampliada asociada al sistema {\bf (2 puntos)}
\[
(A\,|\,b)=\left( 
    \begin{array}{ccc|c}
    1 & 1  & 1 & 4 \\ 2 & 5 & -\alpha & 3\\ 1 & 2 & 1& 8  \end{array}
\right)\]
Haciendo operaciones fila, tenemos que {\bf (5 puntos)}
\[
(A\,|\,b)\sim \left( 
    \begin{array}{ccc|c} 1 & 1  & 1 & 4 \\ 0 & 3 & -\alpha-2 & -5\\ 0 & 0 & -\alpha -2 & -17  \end{array}
\right)\]

Por teorema del rango
\begin{enumerate}
    \item Si $\alpha=-2$, entonces $\operatorname{rank}(A)=2<3=\operatorname{rank}(A|b)$, por lo tanto el sistema no tendr\'ia soluci\'on. {\bf (4 puntos)}
    \item Si $\alpha\neq -2$, entonces $\operatorname{rank}(A)=3=\operatorname{rank}(A|b)$, por lo tanto el sistema tendr\'ia soluci\'on. Como el n\'umero de ecuaciones es 3, tenemos que la soluci\'on es \'unica. {\bf (4 puntos)}
\end{enumerate}
En en caso $\alpha=3$, tenemos que la soluci\'on es \'unica y {\bf (3 puntos)}
\[
(A\,|\,b)\sim \left( 
    \begin{array}{ccc|c} 1 & 0  & 0 & -17/5 \\ 0 & 1 & 0 & 4\\ 0 & 0 & 1 & 17/5  \end{array}
\right)\]
Por lo tanto la soluci\'on es $\{(x,y,z)=(-17/5,4,17/5)\}$. {\bf (2 puntos)}




\end{enumerate}
