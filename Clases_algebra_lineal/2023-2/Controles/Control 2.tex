\section*{Formulario}

\begin{itemize}
    \item {\bf Subespacio vectorial: }Si para todo $\alpha \in K$ y para todo $x, y \in H$, $\alpha x+y \in H$, entonces $H$ es subespacio vectorial.
    \item {\bf Combinación lineal: } Un vector de la forma $v = a_1 v_1 + a_2v_2 + \cdots +a_n v_n$, donde $a_1, a_2, \dots a_n$ son escalares, se denomina una combinación lineal de $v_1, v_2, \dots , v_n$.
\item {\bf Linealmente Dependiente e Independiente: }El conjunto $A=\{v_1, v_2, \dots, v_n\}$ es
Linealmente Independiente (L.I.) si $ a_1 v_1 + a_2v_2 + \cdots +a_n v_n=0$ implica que $a_i=0$ para todo $i=1, \dots, n.$ En caso contrario, es linealmente dependiente (L.D.).
\item {\bf Base: } Un conjunto $\{ v_1,v_2, \dots, v_n \}$ es una base para un espacio vectorial $V$ si $\{ v_1,v_2, \dots, v_n \}$ es linealmente independiente y genera a $V$.

%\item {\bf Proyección ortogonal: }Sea $H$ un subespacio de $\mathbb{R}^n$ con base ortogonal $\{ w_1,w_2, \dots , w_k  \}$. Si $v \in \mathbb{R}^n$, entonces la proyección ortogonal de $v$ sobre $W$, denotada por $P_W(v)$, está dada por:
%$$P_W( v) = \frac{\langle v, w_1 \rangle}{\langle w_1, w_1 \rangle}w_1 + \frac{\langle v, w_2 \rangle}{\langle w_2, w_2 \rangle}w_2 + \cdots + \frac{\langle v, w_k \rangle}{\langle w_k, w_k \rangle}w_k.$$
%Adem\'as, $P_{W^\perp}( v)=v-P_W( v)$
\item {\bf Suma de subespacios: }
Sea $V$ un espacio vectorial sobre $K$ y sean $W_1, W_2$ subespacios de $V$. Definimos la suma de los subespacios $W_i$ y anotamos $W_1 + W_2$ como es subespacio
\[\{v_1+v_2 \colon v_1\in W_1, v_2\in W_2\}.\]

Adem\'as, se cumple que 
\[\dim_K(W_1+ W_2)=\dim_K(W_1)+\dim_K(W_2)-\dim_K(W_1\cap W_2).\]
\item {\bf Transformaciones lineales: } Transformación Lineal $T$ de $ V$ en $W$, como una función $T:V\to W$ tal que
\begin{enumerate}
    \item $T(v+w)=T(v)+T(w)$,
    \item $T(\alpha v)=\alpha T(v)$,
\end{enumerate}
para todo $v,w\in V$, para todo $\alpha \in K$.
\end{itemize}
