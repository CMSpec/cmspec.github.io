\begin{enumerate}
    \item {\bf (12 puntos)} Usando el teorema del rango, determine los valores de $\alpha$ de modo que el sistema
    \begin{align*}
 x + 3y &= 1 \\
 x + 2y +z &= 0\\
2 x + 5y +(\alpha -1) z &= 1-\alpha 
\end{align*}

    tenga solución única, infinitas soluciones o ninguna solución. Para $\alpha=0$, determine el conjunto soluci\'on del sistema.


\newpage 

\item Considere los conjuntos $B_1= \{ \begin{pmatrix} 2 \\ 1 \\ 2 \end{pmatrix}, \begin{pmatrix} 0 \\ 0 \\ 2 \end{pmatrix}, \begin{pmatrix} 1 \\ 0 \\ 2 \end{pmatrix} \}$ y $B_2= \{ \begin{pmatrix} 0 \\ 1 \\ 1 \end{pmatrix}, \begin{pmatrix} 1 \\ 0 \\ 0 \end{pmatrix}, \begin{pmatrix} 0 \\ 0 \\ 1 \end{pmatrix} \} $
\begin{enumerate}
    \item {\bf (6 puntos)} Muestre que tanto $B_1$ como $B_2$ son bases de $\mathbb{R}^3$
    \item {\bf (6 puntos)} Encuentre la matriz de cambio de base de $B_1$ a $B_2$
\end{enumerate}

\newpage

\item {\bf (12 puntos)} Demuestre que esta función $T: \mathbb{R}^{3} \to \mathbb{R}^{3}$ dada por $${T \begin{pmatrix} x \\ y \\ z \end{pmatrix} = \begin{pmatrix}
    2x+z \\ x-y+z \\ y+2z
\end{pmatrix}}$$
 es lineal y encuentre su matriz representante en la base canónica.

\newpage
    \item Sea $H$ el subconjunto de $\mathbb{R}^3$ definido por
\[H=\{(x,y,z)\in\mathbb{R}^3\colon x-y+z=0  \}\]

\begin{enumerate}
    \item {\bf (6 puntos)} Demuestre que $H$ es un subespacio vectorial de $\mathbb{R}^3$.
    \item {\bf (6 puntos)} Encuentre una base de $H$ y determine su dimensión.

\end{enumerate}

\newpage
    \item Considere la siguiente matriz
\[    A=\begin{pmatrix} 1 &0 & 0 \\ 2 & 1 & 4 \\ 1 & 1 & 1 \end{pmatrix}.\]
\begin{enumerate}
    \item {\bf (6 puntos)} Encuentre los valores propios de $A$.
    \item {\bf (6 puntos)} Determine si es diagonalizable, en caso positivo, encuentre la matriz diagonal.
\end{enumerate}
\end{enumerate}
